// test 01 -- conversions between primitive types
routine conversion_test() is

    var true_int   : integer is 1
    var false_int  : integer is 0

    var standard_int   : integer is 5
    var standard_real  : real    is 3.14
    var standard_true  : boolean is true
    var standard_false : boolean is false

    var int_bucket  : integer
    var real_bucket : real
    var bool_bucket : boolean

    // direct copying
    int_bucket  := standard_int
    real_bucket := standard_real
    bool_bucket := standard_true
    print int_bucket, real_bucket, bool_bucket   // 5, 3.14, true

    // integer := real, rounding to nearest
    int_bucket := standard_real
    print int_bucket                              // 3
    int_bucket := 3.7
    print int_bucket                              // 4
    int_bucket := -2.5
    print int_bucket                              // -2

    // integer := boolean
    int_bucket := standard_true
    print int_bucket                              // 1
    int_bucket := standard_false
    print int_bucket                              // 0

    // real := integer
    real_bucket := standard_int
    print real_bucket                             // 5.0

    // real := boolean
    real_bucket := standard_true
    print real_bucket                             // 1.0
    real_bucket := standard_false
    print real_bucket                             // 0.0

    // boolean := integer, only 1 and 0 are legal
    bool_bucket := true_int
    print bool_bucket                             // true
    bool_bucket := false_int
    print bool_bucket                             // false
end
