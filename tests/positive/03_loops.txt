// test 03 -- all loop forms

type IntArr is array [5] integer

routine loops_test() is
    var s : integer
    var i : integer

    // while loop
    s := 0
    i := 1
    while i <= 5 loop
        s := s + i
        i := i + 1
    end
    print s                      // 15

    // for over an ascending range
    s := 0
    for k in 1 .. 5 loop
        s := s + k
    end
    print s                      // 15

    s := 0
    for k in 5 .. 1 loop
        s := s + 1
    end
    print s                      // 0

    for k in 1 .. 3 reverse loop
        print k                  // 3, then 2, then 1
    end

    for k in 4 .. 4 loop
        print k                  // 4
    end

    var n : integer is 3
    for k in 1 .. n loop
        n := 10
        print k                  // 1, 2, 3 
    end

    s := 0
    for a in 1 .. 3 loop
        for b in 1 .. 3 loop
            s := s + 1
        end
    end
    print s                      // 9

    s := 0
    while false loop
        s := 1
    end
    print s                      // 0

    var arr : IntArr
    for k in 1 .. 5 loop
        arr[k] := k * k
    end

    for x in arr loop
        print x                  // 1, 4, 9, 16, 25
    end

    for x in arr reverse loop
        print x                  // 25, 16, 9, 4, 1
    end
end
