// test 03c -- operator precedence
// taken literally, the grammar puts * / % in Simple and + - in Factor,
// which makes multiplication bind LOOSER than addition:
//     2 + 3 * 4  would parse as  (2 + 3) * 4  =  20
// Уточнить 

routine precedence_test() is
    print 2 + 3 * 4              // 14, not 20
    print 10 - 2 * 3             // 4, not 24
    print (2 + 3) * 4            // 20 -- parentheses force the other grouping
    print 1 + 2 < 4              // true -- relations bind looser than arithmetic
    print true and false or true // OPEN: and/or have equal precedence here
end
