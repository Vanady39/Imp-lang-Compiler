// test 05 -- scoping, shadowing and type inference
// expected: compiles, runs, prints the values in the comments

type MyInt is integer

routine shadow_test() is
    var x : integer is 1

    if true then
        var x : integer is 2
        print x                  // 2
    end
    print x                      // 1 -- the inner x is gone

    var t : integer is 5
    var c : integer is 0
    while c < 2 loop
        var t : integer is 99
        print t                  // 99, then 99 again
        c := c + 1
    end
    print t                      // 5

    var i : integer is 100
    for i in 1 .. 3 loop
        print i                  // 1, 2, 3 -- loop variable shadows the outer i
    end
    print i                      // 100
end

routine type_shadow_test() is
    var a : MyInt is 7
    print a                      // 7

    if true then
        type MyInt is real
        var b : MyInt is 2.5
        print b                  // 2.5
    end

    var c : MyInt is 8           // MyInt is integer again out here
    print c                      // 8
end

routine synonym_test() is
    var a : MyInt   is 5
    var b : integer is 10

    b := a
    a := b
    print a, b                   // 5, 5
end

routine inference_test() is
    var inferred_one  is 1
    var inferred_real is 3.9
    var inferred_bool is true

    var probe_bool : boolean
    probe_bool := inferred_one
    print probe_bool             // true

    var probe_int : integer
    probe_int := inferred_real
    print probe_int              // 4

    var probe_bool2 : boolean
    probe_bool2 := inferred_bool
    print probe_bool2            // true
end
