// test 05e -- result types of arithmetic

routine arith_inference_test() is
    var mixed is 1 + 2.5         // 3.5
    var div   is 7 / 2           // 3.5
    var rem   is 7 % 2           // 1

    var probe : integer
    probe := mixed
    print probe                  // 4

    print div
    print rem                    // 1

    var cmp is 1 < 2             // boolean
    print cmp                    // true
end
