// test 06 -- unsized array parameter accepts different sizes and preserves aliasing
// entry: array_parameter_test
// expected: COMPILE OK, RUN OK
// expected output (one value per line): 11, 17
// Project I pp. 3-4, 6: sizeless array parameters; passing copies the reference.
// No array-length member or command-line aggregate input is needed.

routine change_first(a : array [] integer) is
    a[1] := a[1] + 10
end

routine array_parameter_test() is
    var small : array [2] integer
    var large : array [4] integer

    small[1] := 1
    large[1] := 7

    change_first(small)
    change_first(large)

    print small[1]              // 11
    print large[1]              // 17
end
